dim x, y, i %
dim rate !
dim flag $
readln x, y
rate := .5E-2
flag := !(x <> y) or true and false
x := 101B + 7O + 10D + 0AFH
if (x >= y) [writeln x: x := x - 1] else writeln y
for i := 1 to 10 step 2 [x := x + i
writeln x] next
for i := 1 to 3 writeln i next
while (x > 0) [x := x - 1: writeln x]
writeln (x + y) * 2 / 3, flag
end
