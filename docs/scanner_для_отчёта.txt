Листинг основной логики лексического анализатора.
Точные методы analyze и scanner из работающей программы.
Цикл analyze соответствует repeat ... until (CS=V) or (CS=ER).
scanner соответствует case CS of, записанному через if/elif.
Служебные функции перечислены в Функции_if_elif.txt.
Номера таблиц: TW=1, TL=2, TI=3, TN=4; отсутствие при look: z=0.
B хранит символы числа; put(4, kind) переводит его в точное значение.

    def analyze(self, source: str, *, trace: bool = True) -> AnalysisResult:
        self._reset(source, trace)
        while self.CS not in ("V", "ER"):
            try:
                self.scanner()
            except NumericLimitError as error:
                self._error(str(error), self.last_edge)
                self.CS = "ER"
        return self.result

    def scanner(self) -> None:

        _record нужен только для координат лексем и журнала переходов.

        if self.CS == 'START':
            # M01: START -> H
            self._record('M01', 'H', 'gc')
            self.gc()
            self.CS = 'H'

        elif self.CS == 'H':
            if self.CH != EOF and self.CH in ' \t':
                # M02: H -> H
                self._record('M02', 'H', 'gc')
                self.gc()

            elif let(self.CH):
                # M03: H -> I
                self._record('M03', 'I', 'nill, add, gc')
                self.nill()
                self.add()
                self.gc()
                self.CS = 'I'
            elif self.CH != EOF and self.CH in '01':
                # M04: H -> N2
                self._record('M04', 'N2', 'B_start, gc')
                self.B = self.CH
                self.gc()
                self.CS = 'N2'
            elif self.CH != EOF and self.CH in '234567':

                self._record('M06', 'N10', 'B_start, gc')
                self.B = self.CH
                self.gc()
                self.CS = 'N10'
            elif self.CH != EOF and self.CH in '.':
                # M07: H -> T
                self._record('M07', 'T', 'B_reset, B_append, gc')
                self.B = ""
                self.CS = 'T'
                self.B += self.CH
                self.gc()
            elif self.CH != EOF and self.CH in '{':
                # M08: H -> C
                self._record('M08', 'C', 'gc')
                self.gc()
                self.CS = 'C'
            elif self.CH != EOF and self.CH in ':':

                self._record('M09', 'DV', 'gc')
                self.gc()
                self.CS = 'DV'

                self.gc()
                self.CS = 'CM'

                self._record('M11', 'CB', 'gc')

                self.CS = 'CB'
